if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $nama  = filter_input(INPUT_POST, 'nama', FILTER_SANITIZE_SPECIAL_CHARS);
    $email = filter_input(INPUT_POST, 'email', FILTER_VALIDATE_EMAIL);
    $pesan = filter_input(INPUT_POST, 'pesan', FILTER_SANITIZE_SPECIAL_CHARS);

    if (!$nama || !$email || !$pesan) {
        echo json_encode(['status' => 'error', 'message' => 'Harap isi semua kolom dengan benar.']);
        exit;
    }

    // Tautkan pesan langsung ke WhatsApp Alfath Adi Putra
    $waNumber = "6285935617640";
    $textMsg  = "Halo Alfath Adi Putra,\n\nNama: {$nama}\nEmail: {$email}\nPesan: {$pesan}";
    $waUrl    = "https://wa.me/{$waNumber}?text=" . urlencode($textMsg);

    echo json_encode([
        'status'   => 'success',
        'message'  => 'Pesan berhasil diproses!',
        'redirect' => $waUrl
    ]);
    exit;
}
